# Opens the panels that are closed by default, from the View menu, and
# records each: the action editor, special functions, chapters and metadata.
# View menu rows at 1600x900: Toolbar 42, Statistics 64, Undo 86, Simulator
# 108, Metadata 130, Action editor 152, Special functions 174, Websocket 196,
# Chapters 218.
#! covers src/UI/OFS_FunscriptMetadataEditor.* src/OpenFunscripter.cpp:ShowMainMenuBar src/OpenFunscripter.cpp:ShowStatisticsWindow
size 1600 900
wait 20
open $MEDIA
wait 120
pause
seek 5
wait 10

lanemove 4.0 10

lanemove 4.5 90

lanemove 5.0 10

tap

tap
