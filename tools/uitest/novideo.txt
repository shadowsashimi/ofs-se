# A script with no video: the startup button, the blank timeline's clock,
# saving and reopening it, and switching to a real video afterwards.
#! covers src/OFS_Project.* src/state/ProjectState.h OFS-lib/videoplayer/* OFS-lib/OFS_MpvLoader.* src/OpenFunscripter.cpp:openFile src/OpenFunscripter.cpp:initProject src/OpenFunscripter.cpp:createStandaloneProject src/OpenFunscripter.cpp:startStandaloneProject src/OpenFunscripter.cpp:ShowNewScriptWindow
size 1600 900
wait 30
shot 00-empty
windows 00-empty
state 00-empty
expect project_valid 0
# Nothing open, so nothing has played.
expect player_time 0.000

# The startup button opens the length dialog.
click 655 388
wait 10
shot 00-dialog
windows 00-dialog
click 798 514
wait 10
windows 00-cancelled

newblank 90 $OUT\novideo.funscript
wait 30
shot 01-blank
state 01-blank
expect project_valid 1
expect standalone 1
expect blank 1
expect duration 90.000
expect paused 1
expect player_time 0.000

# The clock runs on its own and stops when paused.
play
waituntil player_time >= 1.0 300
pause
wait 10
state 02-played
seek 30
wait 5
expect player_time 30.000
expect paused 1

# A point placed on the blank timeline, saved, and read back from disk.
action action_50
wait 5
expect actions 1
action save_project
wait 5

open $OUT\novideo.ofsp
wait 30
state 03-reopened
expect project_valid 1
expect standalone 1
expect blank 1
expect duration 90.000
expect actions 1

# A real video afterwards replaces the blank timeline entirely.
open $MEDIA
wait 120
state 04-media
expect standalone 0
expect blank 0
waituntil duration >= 59.9 60
# A video is never mistaken for audio alone.
expect audio_only 0
quit
