# With a second lane ticked, Select All and the move keys act on both lanes,
# so a multi-axis script can be shifted in time as one, and one undo puts
# both back.
#! covers OFS-lib/Funscript/Funscript.cpp:MoveSelectionTime OFS-lib/Funscript/Funscript.cpp:LimitSelectionTimeOffset src/OpenFunscripter.cpp:moveSelectionTime src/OpenFunscripter.cpp:moveSelectionPosition src/OpenFunscripter.cpp:selectAll src/OpenFunscripter.cpp:clearSelection src/OpenFunscripter.cpp:selectTime
